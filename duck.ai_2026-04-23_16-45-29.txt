import numpy as np

def generate_synthetic_patient_data(n_patients=100):
    """Generate fake patient data for hackathon"""

    for patient_id in range(n_patients):
        # Generate 24 hours of vital signs (hourly)

            # Stable patient

                hr = np.random.normal(75, 5)  # Normal HR

                hr = np.random.normal(100, 10)  # Rising HR

                temp = np.random.normal(38.5, 0.5)  # Fever
                deterioration = 1
            
            data.append({
                'patient_id': patient_id,
                'hour': hour,
                'heart_rate': max(40, min(150, hr)),
                'oxygen_sat': max(70, min(100, o2)),
                'systolic_bp': max(80, min(180, bp_sys)),
                'temperature': temp,
                'deterioration_label': deterioration
            })
    
    return pd.DataFrame(data)

# Generate 100 synthetic patients
